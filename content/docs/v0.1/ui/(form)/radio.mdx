<Steps>
  <Step>
    ### Install the following dependencies:

        ```bash
        npm install @radix-ui/react-radio-group
        ```
      </Tab>
      <Tab value="pnpm">
        ```bash
        pnpm add @radix-ui/react-radio-group
        ```
      </Tab>
